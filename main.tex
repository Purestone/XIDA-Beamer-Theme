\begin{frame}{Mathematical expressions}
  Inline notation: $x_t \in \mathbb{R}^{d}$, with parameter vector $\theta$.
  \begin{equation}
    \hat{y}_{t+1}=f_{\theta}(x_t), \qquad
    \mathcal{L}(\theta)=\frac{1}{n}\sum_{i=1}^{n}(y_i-\hat{y}_i)^2.
  \end{equation}
  \begin{block}{Definition}
    The loss measures mean squared prediction error.
  \end{block}
\end{frame}
\begin{frame}{A theorem and a proof}
  \begin{theorem}[Cauchy--Schwarz inequality]
    For vectors $x,y\in\mathbb{R}^{n}$,
    \[
      |x^{\mathsf T}y|\leq \lVert x\rVert_2\lVert y\rVert_2.
    \]
  \end{theorem}
  \begin{proof}
    The result follows from the non-negativity of $\lVert x-ty\rVert_2^2$.
  \end{proof}
\end{frame}
